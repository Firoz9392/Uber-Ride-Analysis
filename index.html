SQL analysis

Queries behind the dashboard

Monthly revenue with a window comparison
WITH monthly_revenue AS (
  SELECT DATE_TRUNC('month', booking_date) AS month,
         SUM(fare) AS revenue
  FROM rides WHERE ride_status = 'Completed'
  GROUP BY 1
)
SELECT month, revenue, LAG(revenue) OVER (ORDER BY month) AS previous_revenue
FROM monthly_revenue ORDER BY month;
Driver ranking by completed-ride revenue
SELECT d.driver_name, RANK() OVER (ORDER BY SUM(r.fare) DESC) AS revenue_rank,
       COUNT(*) AS completed_rides, SUM(r.fare) AS revenue
FROM rides r JOIN drivers d ON d.driver_id = r.driver_id
WHERE r.ride_status = 'Completed'
GROUP BY d.driver_name;
